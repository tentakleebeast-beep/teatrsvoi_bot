# Публичные маршруты tool-api: пометка «публичный» и причина (docs/development/26 §7c).
#
# Маршрут считается защищённым, если в его подписи есть Depends(...)/Header(...).
# Всё остальное обязано быть в этом реестре: новая публичная ручка без строки здесь —
# находка arch_drift.py («публичный маршрут без пометки»). Мёртвая строка (маршрута
# больше нет) — тоже находка: реестр не должен гнить.
#
# Формат: <путь> # <причина>
#
# Реестр ничего не разрешает в коде: он фиксирует уже принятое решение, чтобы оно
# было видно числом. Открытие нового публичного маршрута — решение владельца.

# --- оболочка Mini App: HTML без данных, данные приходят через защищённые /webapp/* ---
/webapp # анкета участника, открывается кнопкой web_app из бота

/webapp/week # вкладка «Время»: только разметка
/webapp/week/summary # сводка занятости: только разметка
/webapp/week/month # сетка месяца «люди × дни»: только разметка, данные через /webapp/month-grid
/webapp/notes # экран заметок: только разметка, данные через /webapp/notes/list
/webapp/boot # маяк загрузки Mini App: только телеметрия (длина initData, платформа, JS-ошибка), данных не отдаёт и не хранит
/webapp/money # вкладка «Деньги»: только разметка
/webapp/tx # форма операции: только разметка
/webapp/dues # взносы: только разметка
/webapp/notifications # уведомления: только разметка
/webapp/feedback # обратная связь: только разметка
/webapp/slots # слоты: только разметка
/webapp/reconcile # сверка: только разметка, данные через защищённый API
/webapp/broadcast # рассылка: только разметка, права проверяет API
/webapp/ui # витрина элементов интерфейса: только разметка
/webapp/services # услуги: только разметка
/webapp/files # файлы: только разметка
/webapp/folder # папка: только разметка
/webapp/person # человек: только разметка
/webapp/admin # админ-панель: только разметка, данные через защищённый API
/webapp/search # поиск: только разметка
/webapp/days # дни: только разметка
/webapp/my/sheet # моя занятость: только разметка
/webapp/schedule # расписание: только разметка
